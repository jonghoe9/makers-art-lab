## 3. 스위치 입력 
```cpp title="ew_0603.ino" linenums="1" hl_lines="9"
//
// Program #3. Digital Input
//

#define LED_PIN 13
#define SW_PIN  8

void setup() {
  pinMode(LED_PIN, OUTPUT);
}

void loop() {
  bool status = LOW;
  status = digitalRead(SW_PIN);
  if(status == HIGH) {
    digitalWrite(LED_PIN, HIGH);
  } else {
    digitalWrite(LED_PIN, LOW);
  }
}
```

## 6. 시리얼 입력
```cpp title="ew_0606.ino" linenums="1" hl_lines="9"
//
// Program #6. Serial Input
//

#define LED_PIN 9


void setup() {
  pinMode(LED_PIN, OUTPUT);
  Serial.begin(115200);
}

void loop() {
  while(Serial.available() > 0) {
    int data = Serial.parseInt();
    int led_val = constrain(data, 0, 255);
    if(Serial.read() == '\n') {
      analogWrite(LED_PIN, led_val);
      Serial.print(data);
      Serial.print(" : ");
      Serial.println(led_val);
    }
  }
}
```


## 7. 시리얼 입력 (프로세싱과 연동)

* 아두이노는 시리얼에서 숫자 2개를 읽음
* 숫자 2개는 서보모터 2개의 위치 값으로 사용

```cpp title="ew_0607.ino" linenums="1" hl_lines="9"
//
// Program #7. Serial Input (Arduino)
//
#include <Servo.h>

#define LED_PIN 9
#define SV1_PIN 10

Servo sv1;

void setup() {
  Serial.begin(115200);
  sv1.attach(SV1_PIN);
}

void loop() {
  while(Serial.available() > 0) {
    int data1 = Serial.parseInt();
    int data2 = Serial.parseInt();
    int sv1_val = map(data1, 0, 800, 0, 180);
    int sv2_val = map(data2, 0, 500, 0, 180);

    if(Serial.read() == '\n') {
      sv1.write(sv1_val);
      Serial.print(data1);
      Serial.print(":");
      Serial.print(led_val);
      Serial.print("_");
      Serial.print(data2);
      Serial.print(":");
      Serial.println(sv1_val);
    }
  }
}
```

* 프로세싱은 시리얼로 숫자 2개를 출력함
* 여기서는 숫자 2개를 마우스의 위치값 x, y 를 사용함

```java title="ew_0607_processing.pde" linenums="1"
//
// // Program #7. Serial Write (Processing)
//
import processing.serial.*;

Serial port;

  size(800, 500);

  background(255);
  line(0, mouseY, width, mouseY);

  port.write(mouseX + ", " + mouseY + '\n');
}
```
