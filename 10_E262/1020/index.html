    #include <Adafruit_NeoPixel.h>

#define PIXEL_PIN       6           // 아두이노에서 NeoPixel 연결하는 핀
#define PIXEL_CNT       4           // NeoPixel 수

Adafruit_NeoPixel pixels(PIXEL_CNT, PIXEL_PIN, NEO_GRB + NEO_KHZ800);
                                    // NeoPixel Class 연결

void setup() {
  pixels.begin();
}

void loop() {
  pixels.clear();
  pixels.setPixelColor(0, pixels.Color(150, 0, 0));
  pixels.setPixelColor(1, pixels.Color(0, 150, 0));
  pixels.setPixelColor(2, pixels.Color(0, 0, 150));
  pixels.setPixelColor(3, pixels.Color(100, 150, 250));
  pixels.show();
}

    Mad Mapper와 NeoPixel 연동하기

    • MadMapper는 시리얼 출력이 없음
    • MadMapper에서 조명기(DMX-Fixture) 사용해 출력
    • 조명기(Fixture)로 LED 지정, LED 숫자와 배치 방법 정하기
    • DMX 채널 설정 후 Artnet 으로 출력 지정
    • 색깔 변화를 주면 DMX 신호로 바뀌어 출력됨
    • DMX 신호는 DMX 인터페이스 혹은 Artnet 단자로 출력됨
    • DMX 또는 Artnet 신호를 시리얼로 바꾸는 작업 필요
    • Artnet을 시리얼로 바꾸기는 TouchDesigner / Processing 에서 가능

#define PIXEL_PIN       6           // 아두이노에서 NeoPixel 연결하는 핀
#define PIXEL_CNT       4           // NeoPixel 수

Adafruit_NeoPixel pixels(PIXEL_CNT, PIXEL_PIN, NEO_GRB + NEO_KHZ800);
                                    // NeoPixel Class 연결

void setup() {
  Serial.begin(115200);
  pixels.begin();
}

void loop() {
  while(Serial.available() > 0) {
    int data1 = Serial.parseInt();
    int data2 = Serial.parseInt();
    int data3 = Serial.parseInt();
    int data4 = Serial.parseInt();
    if(Serial.read() == '\n') {
      pixels.clear();
      pixels.setPixelColor(data1, pixels.Color(data2, data3, data4));
      pixels.show();
    }
  }
}

    #include <Adafruit_NeoPixel.h>

#define PIXEL_PIN       6           // 아두이노에서 NeoPixel 연결하는 핀
#define PIXEL_CNT       4           // NeoPixel 수
Adafruit_NeoPixel pixels(PIXEL_CNT, PIXEL_PIN, NEO_GRB + NEO_KHZ800);
int pixel_color[PIXEL_CNT *3];

void setup() {
  Serial.begin(115200);
  pixels.begin();
  for(int i=0; i < PIXEL_CNT *3; i++) {
    pixel_color[i] = 0;
  }
}

void loop() {
  while(Serial.available() > 0) {
    int data1 = Serial.parseInt();
    int data2 = Serial.parseInt();
    int data3 = Serial.parseInt();
    int data4 = Serial.parseInt();

    if(Serial.read() == '\n') {
      int idx_c = data1*3;
      pixel_color[idx_c+0] = data2;
      pixel_color[idx_c+1] = data3;
      pixel_color[idx_c+2] = data4;
    }

    pixels.clear();
    int idx = 0;
    for(int i=0; i < PIXEL_CNT; i++) {
      idx = i;
      int idx_c=idx*3;
      pixels.setPixelColor(idx, pixels.Color(pixel_color[idx_c+0], pixel_color[idx_c+1], pixel_color[idx_c+2]));
    }
    pixels.show();
  }
}

    #include <Adafruit_NeoPixel.h>

#define PIXEL_PIN 6 
#define PIXEL_CNT 4 
#define TOTAL_CHANNELS (PIXEL_CNT * 3)

Adafruit_NeoPixel pixels(PIXEL_CNT, PIXEL_PIN, NEO_GRB + NEO_KHZ800);
int pixel_color[TOTAL_CHANNELS];

void setup() {
  Serial.begin(115200);
  pixels.begin();
}

void loop() {
  if (Serial.available() > 0) {
    int temp_color[TOTAL_CHANNELS];
    int count = 0;
    bool has_newline = false;
    // 패킷의 끝(\n)을 만나거나 버퍼가 빌 때까지 숫자를 계속 읽음
    while (Serial.available() > 0 && !has_newline) {
      // 다음 읽을 데이터가 숫자가 아닌 줄바꿈(\n)이나 공백인 경우 체크
      char next_char = Serial.peek();
      if (next_char == '\n') {
        has_newline = true;
        Serial.read(); // \n 버퍼에서 제거
        break;
      }
      // 숫자라면 파싱하여 임시 배열에 저장 (12개까지만 저장)
      if (next_char >= '0' && next_char <= '9') {
        int val = Serial.parseInt();
        if (count < TOTAL_CHANNELS) {
          temp_color[count] = val;
        }
        count++;
      } else {
        // 공백, 컴마, 의미없는 문자는 그냥 읽어서 버림
        Serial.read(); 
      }
    }
    // 줄바꿈(\n)이 정상적으로 감지되었고, 그 전까지 센 숫자가 정확히 12개인 경우
    if (has_newline && count == TOTAL_CHANNELS) {
      for (int i = 0; i < TOTAL_CHANNELS; i++) pixel_color[i] = temp_color[i];
      // 네오픽셀 출력
      for (int i = 0; i < PIXEL_CNT; i++) {
        int idx = i * 3;
        pixels.setPixelColor(i, pixels.Color(pixel_color[idx], pixel_color[idx+1], pixel_color[idx+2]));
      }
      pixels.show();
    }
  }
}

#include <Adafruit_NeoPixel.h>

#define PIXEL_PIN 6
#define PIXEL_CNT 4
#define CHANNEL_CNT (PIXEL_CNT * 3)
#define BAUD_RATE 115200

Adafruit_NeoPixel pixels(
  PIXEL_CNT, PIXEL_PIN, NEO_GRB + NEO_KHZ800
);

uint8_t pixelColor[CHANNEL_CNT] = {0};

char serialBuffer[64];
uint8_t bufferIndex = 0;

void setup() {
  Serial.begin(BAUD_RATE);

  pixels.begin();
  pixels.clear();
  pixels.show();
}

void loop() {
  // 수신된 데이터를 한 글자씩 처리
  while (Serial.available() > 0) {
    char c = Serial.read();

    if (c == '\n') {
      serialBuffer[bufferIndex] = '\0';

      if (parseRGB(serialBuffer)) {
        updatePixels();
      }

      bufferIndex = 0;
    }
    else if (c != '\r') {
      if (bufferIndex < sizeof(serialBuffer) - 1) {
        serialBuffer[bufferIndex++] = c;
      } else {
        bufferIndex = 0;
      }
    }
  }
}

// RGB 12개 값 파싱
bool parseRGB(char* input) {
  uint8_t values[CHANNEL_CNT];
  char* ptr = input;

  for (int i = 0; i < CHANNEL_CNT; i++) {
    while (*ptr == ' ' || *ptr == ',' || *ptr == '\t') {
      ptr++;
    }

    if (*ptr < '0' || *ptr > '9') {
      return false;
    }

    char* endPtr;
    long value = strtol(ptr, &endPtr, 10);

    if (value < 0 || value > 255) {
      return false;
    }

    values[i] = (uint8_t)value;
    ptr = endPtr;

    if (i < CHANNEL_CNT - 1) {
      if (*ptr != ',' && *ptr != ' ' && *ptr != '\t') {
        return false;
      }
    }
  }

  while (*ptr == ' ' || *ptr == '\t') {
    ptr++;
  }

  if (*ptr != '\0') {
    return false;
  }

  memcpy(pixelColor, values, CHANNEL_CNT);
  return true;
}

// 변경된 RGB 데이터를 LED에 적용
void updatePixels() {
  for (int i = 0; i < PIXEL_CNT; i++) {
    int idx = i * 3;

    pixels.setPixelColor(
      i,
      pixelColor[idx],
      pixelColor[idx + 1],
      pixelColor[idx + 2]
    );
  }

  pixels.show();
}
